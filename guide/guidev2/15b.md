# part 15b | edit form — simple file upload

> **PATH: simple only.** Add these deltas to `Edit{{resource}}Form` (`15.md` step 3) and `AdminEdit{{resource}}Form` (`15.md` step 5).

## user edit form (step 4 delta)

- A: `FileDropzone` import (`useState` already imported)
- B: file state variable
- C: update `handleSubmit` to pass file
- D: input for file upload

A:

```tsx
import { FileDropzone } from "@/components/ui/file-dropzone";
```

B:

```tsx
const [selectedFile, setSelectedFile] = useState<File | null>(null);
```

C:

```tsx
async function handleSubmit(data: Update{{resource}}Input) {
  setIsSubmitting(true);
  try {
    await update{{resource}}Mutation.mutateAsync({
      id: {{resource}}Data.id,
      data,
      file: selectedFile ?? undefined,
    });
    setIsSubmitting(false);
    form.reset();
    if (!isAlwaysOpen) setIsOpen(false);
    onSuccess?.();
  } catch (err: any) {
    onError?.(err);
    setIsSubmitting(false);
  }
}
```

D:

```tsx
{/* file upload */}
<div className="space-y-2">
  <Label className="text-sm">Featured Image (Optional)</Label>
  <FileDropzone
    preset="{{resource}}Image"
    onFileSelect={setSelectedFile}
    disabled={isSubmitting}
    preview
    currentImageUrl={{{resource}}Data.imagePath ?? undefined}
  />
</div>
```

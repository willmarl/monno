> **PATH: simple only.** Add these deltas to `Create{{resource}}Form` from `14.md` step 2.

- A: `FileDropzone` import (`useState` already imported)
- B: file state variable
- C: update `handleSubmit` to pass file
- D: input for file upload
- E: clear file on reset

A:

```tsx
import { FileDropzone } from "@/components/ui/file-dropzone";
```

B:

```tsx
const [selectedFile, setSelectedFile] = useState<File | null>(null);
```

C:

```tsx
async function handleSubmit(data: Create{{resource}}Input) {
  setIsSubmitting(true);
  try {
    const {{resource}} = await create{{resource}}Mutation.mutateAsync({ data, file: selectedFile ?? undefined });
    handleReset();
    if (!isAlwaysOpen) setIsOpen(false);
    router.push(`/{{resource}}/${{{resource}}.id}`);
    onSuccess?.();
  } catch (err: any) {
    onError?.(err);
    setIsSubmitting(false);
  }
}
```


E — add `setSelectedFile(null)` to `handleReset`:

```tsx
function handleReset() {
  setSelectedFile(null);
```
